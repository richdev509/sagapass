# Isolé du reste du projet Laravel (même principe que scripts/niu-lookup/ et
# scripts/tron-verify/ côté SwapLajan) — dépendances lourdes, jamais mêlées au
# reste. Installer dans un venv dédié (voir README.md de ce dossier).
#
# deepface : correspondance visage pièce <-> selfie, ET détection de vivacité
# intégrée (anti_spoofing=True) — pas besoin d'une librairie anti-spoofing
# séparée (Silent-Face-Anti-Spoofing), DeepFace embarque déjà un modèle
# équivalent (MiniFASNet) pour ça depuis sa version 0.0.90+.
deepface>=0.0.93

# easyocr : extraction des champs texte de la pièce (numéro, nom, date de
# naissance) — licence Apache 2.0, moins de dépendances système que
# PaddleOCR, correct sur des photos de carte prises au téléphone (pas
# seulement des scans plats).
easyocr>=1.7

# "contrib" (pas juste opencv-python-headless) : le modèle SFace utilisé pour
# la correspondance visage (voir check_face_match, model_name="SFace") a
# besoin de cv2.FaceRecognizerSF, qui vit dans les modules "contrib" d'OpenCV
# — confirmé en échouant en conditions réelles avec opencv-python-headless
# seul ("This is an optional dependency. You can install it as pip install
# opencv-contrib-python."). Toujours "headless" (pas de GUI/X11 sur un
# serveur sans écran) et plafonné à <5 (la branche 5.x a retiré les fichiers
# de cascades Haar cv2/data/*.xml qu'attend le détecteur de visage par défaut
# de DeepFace — confirmé séparément en échouant avec la 5.0).
opencv-contrib-python-headless>=4.9,<5

# mediapipe : landmarks faciaux (FaceMesh, TFLite — léger) pour la vivacité
# active (rotation de tête gauche/centre/droite) ET la détection de main sur
# le visage (flux session partenaire QR). Volontairement pas de modèle
# TensorFlow additionnel lourd — voir l'incident OOM avec VGG-Face plus haut,
# même contrainte RAM ici. Plafonné à <1 : la branche 1.x a retiré l'API
# legacy mediapipe.solutions (face_mesh/hands) utilisée ici — confirmé en
# échouant en conditions réelles avec mediapipe 1.0.1 ("module 'mediapipe'
# has no attribute 'solutions'").
mediapipe>=0.10,<1
